using Avalonia.Controls;
using Avalonia.Controls.ApplicationLifetimes;
using Avalonia.Media;
using Avalonia.VisualTree;
using System;
using System.IO;
using System.Linq;
using System.Globalization;
using System.Text.Json;
using System.Collections.Generic;
using System.Threading.Tasks;
internal static class MetadataRuntimeProbe
{
 public static void Attach(IClassicDesktopStyleApplicationLifetime desktop,Window window){window.ShowInTaskbar=false;window.ShowActivated=false;window.Opened+=async(_,_)=>{
  try{await Task.Delay(120);window.UpdateLayout();var rows=new List<object>();using var input=JsonDocument.Parse(File.ReadAllText(Environment.GetEnvironmentVariable("UNIVERSAL5E_RUNTIME_EXPECTATIONS")!));
   foreach(var model in input.RootElement.EnumerateArray()){
    var name=model.GetProperty("Name").GetString();var type=model.GetProperty("Type").GetString();var control=window.GetVisualDescendants().OfType<Control>().Single(c=>c.Name==name&&c.GetType().FullName==type);
    foreach(var p in model.GetProperty("Properties").EnumerateObject()){
     var expected=p.Value.GetString()!;object? value=p.Name=="ToolTip.Tip"?ToolTip.GetTip(control):control.GetType().GetProperty(p.Name)!.GetValue(control);
     var text=value switch{null=>"(null)",bool b=>b?"True":"False",TextTrimming t when ReferenceEquals(t,TextTrimming.None)=>"None",TextTrimming t when ReferenceEquals(t,TextTrimming.CharacterEllipsis)=>"CharacterEllipsis",TextTrimming t when ReferenceEquals(t,TextTrimming.WordEllipsis)=>"WordEllipsis",BoxShadows b=>b.ToString(),double d=>d.ToString("R",CultureInfo.InvariantCulture),Enum e=>e.ToString(),string s=>s,int n=>n.ToString(CultureInfo.InvariantCulture),_=>throw new Exception("Unreviewed runtime value "+p.Name)};
     if(value is BoxShadows)expected=BoxShadows.Parse(expected).ToString();
     if(text!=expected)throw new Exception(type+"."+p.Name+": "+text+" != "+expected);
     rows.Add(new{Type=type,Key=p.Name,Runtime=true,Value=text});
    }
   }
   var result=Environment.GetEnvironmentVariable("METADATA_RUNTIME_RESULT")!;File.WriteAllText(Path.ChangeExtension(result,"properties.json"),JsonSerializer.Serialize(rows));File.WriteAllText(result,"PASS native property pairs="+rows.Count+"; native types=8; trusted native getters only");
  }catch(Exception e){File.WriteAllText(Environment.GetEnvironmentVariable("METADATA_RUNTIME_RESULT")!,"FAIL "+e);}finally{window.Close();desktop.Shutdown();}
 };}
}
